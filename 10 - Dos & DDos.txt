from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from threading import BoundedSemaphore
import time

# Allow only 2 requests to be processed simultaneously
capacity = BoundedSemaphore(3)


class LimitedServer(BaseHTTPRequestHandler):

    def do_GET(self):
        if not capacity.acquire(blocking=False):
            self.send_response(503)
            self.end_headers()
            self.wfile.write(b"Server overloaded! Try again later.")
            print("Request rejected: Server capacity exceeded")
            return

        try:
            print("Request accepted:", self.client_address[0])

            # Simulate slow request processing
            time.sleep(2)

            message = b"Request processed successfully"

            self.send_response(200)
            self.send_header("Content-Length", str(len(message)))
            self.end_headers()
            self.wfile.write(message)

        finally:
            capacity.release()


class Server(ThreadingHTTPServer):
    request_queue_size = 50


server = Server(("0.0.0.0", 8000), LimitedServer)

print("Limited-capacity server running on port 8000")
print("Maximum simultaneous requests: 2")

server.serve_forever()
